cmake_minimum_required(VERSION 3.22.1)
project(shantu_cad C)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LIBREDWG_LIBONLY ON CACHE BOOL "" FORCE)
set(LIBREDWG_DISABLE_JSON ON CACHE BOOL "" FORCE)
set(ENABLE_LTO OFF CACHE BOOL "" FORCE)
set(DISABLE_WERROR ON CACHE BOOL "" FORCE)
set(CAD_SOURCE "${CMAKE_CURRENT_BINARY_DIR}/libredwg-source")
if(NOT EXISTS "${CAD_SOURCE}/CMakeLists.txt")
  file(MAKE_DIRECTORY "${CAD_SOURCE}")
  file(ARCHIVE_EXTRACT INPUT "${CMAKE_CURRENT_SOURCE_DIR}/../../../third-party/libredwg-b39eb1c.tar.gz" DESTINATION "${CAD_SOURCE}")
endif()
add_subdirectory("${CAD_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/libredwg-build")
add_library(shantu_cad SHARED shantu_cad.c)
target_include_directories(shantu_cad PRIVATE "${CAD_SOURCE}/include" "${CAD_SOURCE}/src" "${CMAKE_CURRENT_BINARY_DIR}/libredwg-build/src")
target_link_libraries(shantu_cad redwg m)
target_link_options(shantu_cad PRIVATE "-Wl,-z,max-page-size=16384")
